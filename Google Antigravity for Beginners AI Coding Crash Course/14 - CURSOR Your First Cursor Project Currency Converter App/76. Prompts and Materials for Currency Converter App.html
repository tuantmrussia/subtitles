    export type CurrencyCode = 'USD'|'EUR'|'GBP'|'NPR'|'INR'|'JPY'|'AUD'|'CAD';

    export const CURRENCIES = [

      { code: 'USD', name: 'US Dollar', flag: '🇺🇸' },

      { code: 'NPR', name: 'Nepalese Rupee', flag: '🇳🇵' },

- `data/rates.ts`

  - Mock rates relative to USD and cross-convert:

    ```ts

    import { CurrencyCode } from './currencies';

    export const USD_RATES: Record<CurrencyCode, number> = {

      USD: 1,

      EUR: 0.92,

      GBP: 0.80,

      NPR: 118.16,

      INR: 83.10,

      JPY: 151.2,

      AUD: 1.53,

      CAD: 1.37,

    };

    export function getRate(from: CurrencyCode, to: CurrencyCode): number {

      // Cross via USD table

      const toUsd = 1 / USD_RATES[from];

      return toUsd * USD_RATES[to];

    }

    export function convert(amount: number, from: CurrencyCode, to: CurrencyCode) {

      const rate = getRate(from, to);

      return { rate, total: amount * rate };

    }

    ```
